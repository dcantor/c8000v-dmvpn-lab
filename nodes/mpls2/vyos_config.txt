# mpls2: its own AS, eBGP with every site through a listen range; it carries the sites' WAN addresses and nothing else
set policy prefix-list WAN-ADDRESSES description 'the access links, and nothing else'
set policy prefix-list WAN-ADDRESSES rule 10 action permit
set policy prefix-list WAN-ADDRESSES rule 10 prefix 100.71.0.0/16
set policy prefix-list WAN-ADDRESSES rule 10 le 32
set protocols bgp system-as 65010
set protocols bgp parameters router-id 10.255.5.253
set protocols bgp parameters log-neighbor-changes
set protocols bgp timers keepalive 3
set protocols bgp timers holdtime 9
set protocols bgp peer-group CE remote-as 65100
set protocols bgp peer-group CE description 'the customer sites (AS 65100)'
set protocols bgp peer-group CE address-family ipv4-unicast as-override
set protocols bgp peer-group CE address-family ipv4-unicast prefix-list export WAN-ADDRESSES
set protocols bgp peer-group CE address-family ipv4-unicast prefix-list import WAN-ADDRESSES
set protocols bgp listen limit 64
set protocols bgp listen range 100.71.0.0/16 peer-group CE
